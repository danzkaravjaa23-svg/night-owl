// Actual Flutter admin screens with synthetic data; no live account/network.
import 'dart:io';
import 'dart:ui' as ui;
import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter/services.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:google_fonts/google_fonts.dart';
import 'package:night_owl_ub/core/theme/app_colors.dart';
import 'package:night_owl_ub/core/theme/app_fonts.dart';
import 'package:night_owl_ub/core/theme/app_theme.dart';
import 'package:night_owl_ub/features/admin/providers/admin_console_provider.dart';
import 'package:night_owl_ub/features/admin/screens/admin_panel_screen.dart';
import 'package:night_owl_ub/features/admin/screens/admin_users_screen.dart';
import 'package:night_owl_ub/features/admin/screens/admin_reports_screen.dart';
import 'package:night_owl_ub/features/admin/screens/admin_audit_screen.dart';
import 'package:night_owl_ub/features/admin/screens/admin_content_screen.dart';
import 'admin_console_fixture.dart';

void main() {
  TestWidgetsFlutterBinding.ensureInitialized();
  setUpAll(() async {
    AppFonts.configure();
    await GoogleFonts.pendingFonts([
      for (final weight in [FontWeight.w400, FontWeight.w500, FontWeight.w600, FontWeight.w700])
        GoogleFonts.inter(fontWeight: weight),
    ]);
    await (FontLoader('MaterialIcons')..addFont(rootBundle.load('fonts/MaterialIcons-Regular.otf'))).load();
  });
  for (final dark in [true, false]) {
    for (final compact in [false, true]) {
      for (final item in <String, Widget>{
        'dashboard': const AdminPanelScreen(),
        'reports': const AdminReportsScreen(),
        'users': const AdminUsersScreen(),
        'content': const AdminContentScreen(),
        'audit': const AdminAuditScreen(),
      }.entries) {
        final label = '${item.key}-${dark ? 'night' : 'day'}-${compact ? '320-large' : '390'}';
        testWidgets('admin visual $label', (tester) async {
          final boundary = GlobalKey();
          final service = AdminConsoleFixtureService();
          tester.view.physicalSize = compact ? const Size(320, 740) : const Size(390, 844);
          tester.view.devicePixelRatio = 1;
          addTearDown(tester.view.resetPhysicalSize);
          addTearDown(tester.view.resetDevicePixelRatio);
          addTearDown(() => AppColors.isDarkMode = true);
          AppColors.isDarkMode = dark;
          await tester.pumpWidget(ProviderScope(overrides: [adminConsoleServiceProvider.overrideWithValue(service)],
            child: MaterialApp(theme: dark ? AppTheme.dark : AppTheme.light,
              builder: (context, child) => MediaQuery(data: MediaQuery.of(context).copyWith(
                padding: const EdgeInsets.only(top: 24, bottom: 24), disableAnimations: true,
                textScaler: TextScaler.linear(compact ? 1.5 : 1)),
                child: RepaintBoundary(key: boundary, child: child!)), home: item.value)));
          await tester.pumpAndSettle();
          await tester.runAsync(() async {
            await GoogleFonts.pendingFonts();
            for (final image in tester.widgetList<Image>(find.byType(Image))) {
              await precacheImage(image.image, boundary.currentContext!);
            }
          });
          await tester.pumpAndSettle();
          expect(tester.takeException(), isNull);
          if (const bool.fromEnvironment('CAPTURE_ADMIN')) {
            await tester.runAsync(() async {
              final image = await (boundary.currentContext!.findRenderObject()! as RenderRepaintBoundary).toImage(pixelRatio: 2);
              try {
                final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
                final directory = Directory('build/audit/2026-11-03/admin-security/visual');
                await directory.create(recursive: true);
                await File('${directory.path}/$label.png').writeAsBytes(bytes!.buffer.asUint8List());
              } finally { image.dispose(); }
            });
          }
          await tester.pumpWidget(const SizedBox());
        });
      }
    }
  }
}

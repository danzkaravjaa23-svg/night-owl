const photoUrl = 'https://fixture.images.test/city.png';

// CachedNetworkImage uses this local file only: no cache database or downloads.
class _LocalImages extends BaseCacheManager {
  @override
  Stream<FileResponse> getFileStream(String url,
      {String? key,
      Map<String, String>? headers,
      bool withProgress = false}) async* {
    if (url != photoUrl) throw StateError('Unexpected image fixture: $url');
    yield FileInfo(
        const LocalFileSystem().file('assets/images/tonight_city.png'),
        FileSource.Cache,
        DateTime(2099),
        url);
  }

  @override
  dynamic noSuchMethod(Invocation invocation) => null;
}

// The map venue's Image.network receives only the same bundled city image.
class _LocalHttp implements HttpClient {
  final Uint8List bytes;
  _LocalHttp(this.bytes);
  @override
  Future<HttpClientRequest> getUrl(Uri url) async {
    if (url.toString() != photoUrl) throw StateError('Unexpected image URL');
    return _LocalRequest(bytes);
  }

  @override
  dynamic noSuchMethod(Invocation invocation) => null;
}

class _LocalRequest implements HttpClientRequest {
  final Uint8List bytes;
  _LocalRequest(this.bytes);
  @override
  HttpHeaders get headers => _LocalHeaders();
  @override
  Future<HttpClientResponse> close() async => _LocalResponse(bytes);
  @override
  dynamic noSuchMethod(Invocation invocation) => null;
}

class _LocalHeaders implements HttpHeaders {
  @override
  void add(String name, Object value, {bool preserveHeaderCase = false}) {}
  @override
  dynamic noSuchMethod(Invocation invocation) => null;
}

class _LocalResponse extends Stream<List<int>> implements HttpClientResponse {
  final Uint8List bytes;
  _LocalResponse(this.bytes);
  @override
  int get statusCode => 200;
  @override
  int get contentLength => bytes.length;
  @override
  HttpClientResponseCompressionState get compressionState =>
      HttpClientResponseCompressionState.notCompressed;
  @override
  StreamSubscription<List<int>> listen(void Function(List<int>)? onData,
          {Function? onError, void Function()? onDone, bool? cancelOnError}) =>
      Stream<List<int>>.value(bytes).listen(onData,
          onError: onError, onDone: onDone, cancelOnError: cancelOnError);
  @override
  dynamic noSuchMethod(Invocation invocation) => null;
}

Future<Uint8List> _pixels(GlobalKey key,
    {ui.ImageByteFormat format = ui.ImageByteFormat.png}) async {
  final boundary =
      key.currentContext!.findRenderObject()! as RenderRepaintBoundary;
  final image = await boundary.toImage(pixelRatio: 2);
  try {
    final bytes = await image.toByteData(format: format);
    return bytes!.buffer.asUint8List();
  } finally {
    image.dispose();
  }
}

Future<String> _capture(GlobalKey key, String path) async {
  final bytes = await _pixels(key);
  final file = File(path);
  await file.parent.create(recursive: true);
  await file.writeAsBytes(bytes);
  return sha256.convert(bytes).toString();
}

void main() {
  TestWidgetsFlutterBinding.ensureInitialized();
  setUpAll(() async {
    AppFonts.configure();
    await (FontLoader('MaterialIcons')
          ..addFont(rootBundle.load('fonts/MaterialIcons-Regular.otf')))
        .load();
  });

  for (final dark in [true, false]) {
    for (final narrow in [false, true]) {
      final name =
          'legend-${dark ? 'night' : 'day'}-${narrow ? '320-large' : '390'}';
      testWidgets(name, (tester) async {
        final capture = GlobalKey();
        final size = Size(narrow ? 320 : 390, 844);
        tester.view.physicalSize = size;
        tester.view.devicePixelRatio = 1;
        addTearDown(tester.view.resetPhysicalSize);
        addTearDown(tester.view.resetDevicePixelRatio);
        AppColors.isDarkMode = dark;
        await tester.pumpWidget(MaterialApp(
          debugShowCheckedModeBanner: false,
          theme: dark ? AppTheme.dark : AppTheme.light,
          home: RepaintBoundary(
            key: capture,
            child: Scaffold(
              body: MediaQuery(
                data: MediaQueryData(
                  size: size,
                  disableAnimations: true,
                  textScaler: TextScaler.linear(narrow ? 1.5 : 1),
                ),
                child: const VenueStoryLegend(),
              ),
            ),
          ),
        ));
        await tester.pumpAndSettle();
        await tester.runAsync(() => GoogleFonts.pendingFonts());
        await tester.pumpAndSettle();
        expect(tester.takeException(), isNull);
        expect(tester.binding.hasScheduledFrame, isFalse);
        await tester.runAsync(() => _capture(capture, '$_output/$name.png'));
        await tester.pumpWidget(const SizedBox());
      });
    }
  }

  testWidgets('24 sampled frames show each actual production emotion moving',
      (tester) async {
    tester.view.physicalSize = const Size(390, 420);
    tester.view.devicePixelRatio = 1;
    addTearDown(tester.view.resetPhysicalSize);
    addTearDown(tester.view.resetDevicePixelRatio);
    AppColors.isDarkMode = true;
    final capture = GlobalKey();
    final badgeKeys = {
      for (final level in VenueStoryLevel.values) level: GlobalKey(),
    };
    final hashes = {
      for (final level in VenueStoryLevel.values) level: <String>{},
    };
    final frameHashes = <String>{};
    final frames = <Map<String, Object>>[];
    final phase = AnimationController(vsync: tester);
    try {
      await tester.pumpWidget(MaterialApp(
        debugShowCheckedModeBanner: false,
        theme: AppTheme.dark,
        home: RepaintBoundary(
          key: capture,
          child: Scaffold(body: _gallery(phase, badgeKeys)),
        ),
      ));
      await tester.runAsync(() => GoogleFonts.pendingFonts());
      await tester.pump();
      for (var frame = 0; frame < 24; frame++) {
        phase.value = frame / 24;
        await tester.pump();
        expect(tester.takeException(), isNull);
        final name = 'frame-${frame.toString().padLeft(2, '0')}.png';
        await tester.runAsync(() async {
          final hash = await _capture(capture, '$_output/frames/$name');
          frameHashes.add(hash);
          frames.add({
            'file': 'frames/$name',
            'phase': frame / 24,
            'time_ms': frame * 250,
            'sha256': hash,
          });
          for (final level in VenueStoryLevel.values) {
            final pixels = await _pixels(badgeKeys[level]!,
                format: ui.ImageByteFormat.rawRgba);
            hashes[level]!.add(sha256.convert(pixels).toString());
          }
        });
      }
      for (final level in VenueStoryLevel.values) {
        expect(hashes[level]!.length, greaterThan(1),
            reason: '${level.name} must visibly change over the cycle.');
      }
      expect(frameHashes.length, greaterThan(1));
      await tester.runAsync(() async {
        await File('$_output/motion-capture.json')
            .writeAsString(const JsonEncoder.withIndent('  ').convert({
          'kind': 'actual-production-widget-review',
          'canvas_logical_pixels': {'width': 390, 'height': 420},
          'pixel_ratio': 2,
          'cycle_ms': 6000,
          'frame_duration_ms': 250,
          'frame_count': 24,
          'claim': 'Animation examples, not live venue activity data.',
          'distinct_frames': frameHashes.length,
          'distinct_badge_frames': {
            for (final entry in hashes.entries)
              entry.key.name: entry.value.length,
          },
          'frames': frames,
        }));
      });
      await tester.pumpWidget(const SizedBox());
    } finally {
      phase.dispose();
    }
  });
}

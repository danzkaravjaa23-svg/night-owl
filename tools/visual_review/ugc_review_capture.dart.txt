class _VisualSession extends PublicationSession {
  @override String get ownerId => '10000000-0000-4000-8000-000000000001';
  @override String get token => 'offline-visual-only';
  @override bool get isCurrent => true;
  @override void close() {}
}
class _VisualReportService extends ReportSubmissionService {
  @override PublicationSession captureSession() => _VisualSession();
  @override Stream<void> get sessionChanges => const Stream.empty();
  @override Future<void> submit(PublicationSession session, {
    required String targetType, required String targetId, required String reason, String? details}) =>
      throw StateError('Visual fixture never sends reports.');
}

void main() {
  TestWidgetsFlutterBinding.ensureInitialized();
  setUpAll(() async {
    AppFonts.configure();
    await GoogleFonts.pendingFonts([
      for (final weight in [FontWeight.w400,FontWeight.w500,FontWeight.w600,FontWeight.w700])
        GoogleFonts.inter(fontWeight: weight),
    ]);
    await (FontLoader('MaterialIcons')..addFont(rootBundle.load('fonts/MaterialIcons-Regular.otf'))).load();
  });
  for (final dark in [true,false]) {
    for (final compact in [false,true]) {
      for (final view in ['queue','decision','message-report','message-report-bottom']) {
        final label='$view-${dark?'night':'day'}-${compact?'320-large':'390'}';
        testWidgets('UGC visual $label',(tester) async {
          final key=GlobalKey();
          final service=AdminConsoleFixtureService();
          tester.view.physicalSize=compact?const Size(320,740):const Size(390,844);
          tester.view.devicePixelRatio=1;
          addTearDown(tester.view.resetPhysicalSize);
          addTearDown(tester.view.resetDevicePixelRatio);
          addTearDown(()=>AppColors.isDarkMode=true);
          AppColors.isDarkMode=dark;
          final report=view.startsWith('message-report');
          await tester.pumpWidget(ProviderScope(overrides:[
            adminConsoleServiceProvider.overrideWithValue(service),
            reportSubmissionServiceProvider.overrideWithValue(_VisualReportService()),
          ],child:MaterialApp(theme:dark?AppTheme.dark:AppTheme.light,
            builder:(context,child)=>MediaQuery(data:MediaQuery.of(context).copyWith(
              padding:const EdgeInsets.only(top:24,bottom:24),disableAnimations:true,
              textScaler:TextScaler.linear(compact?1.5:1)),child:RepaintBoundary(key:key,child:child!)),
            home:report?Scaffold(body:Builder(builder:(context)=>Center(child:TextButton(
              onPressed:()=>showReportSheet(context,targetType:'message',
                targetId:'30000000-0000-4000-8000-000000000001',title:'Сонгосон мессежийг мэдээлэх'),
              child:const Text('Жишээ мессежийг мэдээлэх'))))):const AdminContentScreen())));
          await tester.pumpAndSettle();
          if(report){
            await tester.tap(find.text('Жишээ мессежийг мэдээлэх'));
            await tester.pumpAndSettle();
            if(view.endsWith('bottom')){
              await tester.tap(find.text('Дарамт / доромжлол'));
              await tester.enterText(find.byType(TextField),'Зөвхөн сонгосон мессежийг шалгаж өгнө үү.');
              tester.testTextInput.hide();
              await tester.pumpAndSettle();
              await tester.drag(find.byType(SingleChildScrollView).last,const Offset(0,-700));
              await tester.pumpAndSettle();
            }
          }else if(view=='decision'){
            // Stop only when the actual control can receive a tap. A lazy card
            // can already be built while its bottom action remains off-screen.
            final action=find.text('Шалгаад зөвшөөрөх');
            for(var attempt=0;attempt<18&&action.hitTestable().evaluate().isEmpty;attempt++){
              await tester.drag(find.byType(ListView).first,const Offset(0,-220));
              await tester.pumpAndSettle();
            }
            expect(action.hitTestable(),findsOneWidget,reason:'Review action must be reachable by ordinary scrolling.');
            await tester.tap(action.hitTestable());
            await tester.pumpAndSettle();
            await tester.enterText(find.byType(TextField),'Бүх медиа болон текстийг шалгасан.');
            tester.testTextInput.hide();
            await tester.pumpAndSettle();
          }
          expect(tester.takeException(),isNull);
          expect(service.reviewDecisions,isEmpty,reason:'No approval is submitted for a screenshot.');
          await tester.runAsync(() async {
            final image=await(key.currentContext!.findRenderObject()! as RenderRepaintBoundary).toImage(pixelRatio:2);
            try{
              final bytes=await image.toByteData(format:ui.ImageByteFormat.png);
              final file=File('build/audit/2026-11-03/store-implementation/ugc-visual/$label.png');
              await file.parent.create(recursive:true);
              await file.writeAsBytes(bytes!.buffer.asUint8List());
            }finally{image.dispose();}
          });
          await tester.pumpWidget(const SizedBox());
        });
      }
    }
  }
}

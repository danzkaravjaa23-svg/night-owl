const _caption = 'Найзуудтайгаа өнгөрүүлсэн сайхан үдэш.';
const _rule = 'Байршилтай пост газрын хэмнэлд 7 хоног тооцогдоно. '

Future<List<JsonRow>> _catalog(

      },
    ];

Future<void> _capture(GlobalKey key, String path) async {
  final boundary =
      key.currentContext!.findRenderObject()! as RenderRepaintBoundary;
  final image = await boundary.toImage(pixelRatio: 2);
  try {
    final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
    final file = File(path);
    await file.parent.create(recursive: true);
    await file.writeAsBytes(bytes!.buffer.asUint8List());
  } finally {
    image.dispose();
  }
}

void main() {
  TestWidgetsFlutterBinding.ensureInitialized();
  late Directory temp;
  late PickedMediaFile media;
  setUpAll(() async {
    AppFonts.configure();
    await (FontLoader('MaterialIcons')
          ..addFont(rootBundle.load('fonts/MaterialIcons-Regular.otf')))
        .load();
    temp = await Directory.systemTemp.createTemp('nightowl-post-visual-');
    final bytes = await rootBundle.load('assets/images/tonight_city.png');
    final file = File('${temp.path}/night-out.png');
    await file.writeAsBytes(Uint8List.view(
        bytes.buffer, bytes.offsetInBytes, bytes.lengthInBytes));
    media = PickedMediaFile(null, file.path, false);
  });
  tearDownAll(() async {
    AppColors.isDarkMode = true;
    await temp.delete(recursive: true);
  });

  for (final dark in [true, false]) {
    for (final narrow in [false, true]) {
      final name =
          'post-location-${dark ? 'night' : 'day'}-${narrow ? '320-large' : '390'}';
      testWidgets(name, (tester) async {
        final capture = GlobalKey();
        final size = Size(narrow ? 320 : 390, 844);
        tester.view.physicalSize = size;
        tester.view.devicePixelRatio = 1;
        addTearDown(tester.view.resetPhysicalSize);
        addTearDown(tester.view.resetDevicePixelRatio);
        AppColors.isDarkMode = dark;
        await tester.pumpWidget(ProviderScope(
          child: MaterialApp(
            debugShowCheckedModeBanner: false,
            theme: dark ? AppTheme.dark : AppTheme.light,
            builder: (context, child) => MediaQuery(
              data: MediaQuery.of(context).copyWith(
                size: size,
                disableAnimations: true,
                textScaler: TextScaler.linear(narrow ? 1.5 : 1),
              ),
              child: child!,
            ),
            home: RepaintBoundary(
              key: capture,
              child: CreatePostScreen(
                publicationService: _OfflinePublication(),
                mediaPicker: ({bool multiple = true}) async => [media],
                venueLoader: _catalog,
              ),
            ),
          ),
        ));
        await tester.pumpAndSettle();
        await tester.runAsync(() => GoogleFonts.pendingFonts());
        await tester.pumpAndSettle();
        await tester.runAsync(() => precacheImage(previewImageProvider(media),
            tester.element(find.byType(CreatePostScreen))));
        await tester.tap(find.text('Зураг (10 хүртэл) эсвэл видео сонгох'));
        await tester.runAsync(() async {
          await Future<void>.delayed(const Duration(milliseconds: 100));
        });
        await tester.pumpAndSettle();
        final caption = find.byType(TextField).first;
        await tester.scrollUntilVisible(caption, 160,
            scrollable: find.byType(Scrollable).first);
        await tester.enterText(caption, _caption);
        tester.testTextInput.hide();
        FocusManager.instance.primaryFocus?.unfocus();
        await tester.pumpAndSettle();

        final location = find.byKey(const ValueKey('post-location-picker'));
        await tester.scrollUntilVisible(location, 160,
            scrollable: find.byType(Scrollable).first);
        await tester.tap(location);
        await tester.pumpAndSettle();
        expect(find.text('Постын байршил'), findsOneWidget);
        expect(find.textContaining('Нийтэлснээс хойш 7 хоног'), findsOneWidget);
        await tester.tap(find.text(_venueName));
        await tester.pumpAndSettle();

        // Frame the actual caption/location/helper together after choosing.
        // The composer uses a lazy list, so first scroll upward to mount caption.
        await tester.scrollUntilVisible(caption, -120,
            scrollable: find.byType(Scrollable).first);
        await Scrollable.ensureVisible(tester.element(caption), alignment: .02);
        await tester.pumpAndSettle();
        await tester.runAsync(() => GoogleFonts.pendingFonts());
        await tester.pumpAndSettle();
        expect(find.text(_caption), findsOneWidget);
        expect(find.text(_venueName), findsOneWidget);
        expect(find.text(_rule), findsOneWidget);
        final appBar = tester.getRect(find.byType(AppBar));
        expect(tester.getRect(caption).top, greaterThanOrEqualTo(appBar.bottom));
        expect(tester.getRect(find.text(_rule)).bottom,
            lessThanOrEqualTo(size.height - 8));
        expect(tester.getRect(location).width, greaterThan(200));
        expect(tester.takeException(), isNull,
            reason: 'Actual composer must fit mobile and large-text layouts.');
        await tester.runAsync(() => _capture(capture, '$_output/$name.png'));
        await tester.pumpWidget(const SizedBox());
      });
    }
  }
}
